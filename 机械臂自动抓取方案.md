# 机械臂自动抓取技术方案

## 🎯 概述

机械臂自动抓取是一个完整的视觉引导抓取系统，核心流程：

```
目标检测 → 3D定位 → 坐标转换 → 轨迹规划 → 运动执行 → 夹爪控制
   ↓          ↓          ↓          ↓          ↓          ↓
 "看到"     "在哪"     "坐标转换"   "怎么去"    "动起来"    "抓住"
```

### 1.2 目标检测算法

```python
# ============================================================
# 目标检测模块 - 使用 YOLOv8
# ============================================================

from ultralytics import YOLO
import cv2
import numpy as np

class ObjectDetector:
    """目标检测器"""

    def __init__(self, model_path='yolov8n.pt', conf_threshold=0.5):
        """
        初始化检测器

        Args:
            model_path: 模型路径，可以是：
                - 'yolov8n.pt' (最快，精度一般)
                - 'yolov8s.pt' (平衡)
                - 'yolov8m.pt' (推荐)
                - 'yolov8l.pt' (精度高，较慢)
                - 'yolov8x.pt' (最高精度，最慢)
                - 自训练模型路径
            conf_threshold: 置信度阈值
        """
        self.model = YOLO(model_path)
        self.conf_threshold = conf_threshold

    def detect(self, image, target_classes=None):
        """
        检测图像中的目标

        Args:
            image: BGR格式图像 (OpenCV格式)
            target_classes: 要检测的类别列表，如 ['bottle', 'cup']
                           None表示检测所有类别

        Returns:
            list: 检测结果列表，每个元素包含：
                - bbox: [x1, y1, x2, y2] 边界框
                - center: [cx, cy] 中心点
                - class_name: 类别名称
                - confidence: 置信度
        """
        results = self.model(image, conf=self.conf_threshold)[0]

        detections = []
        for box in results.boxes:
            class_id = int(box.cls[0])
            class_name = self.model.names[class_id]

            # 如果指定了目标类别，只返回匹配的
            if target_classes and class_name not in target_classes:
                continue

            x1, y1, x2, y2 = map(int, box.xyxy[0])
            cx = (x1 + x2) // 2
            cy = (y1 + y2) // 2

            detections.append({
                'bbox': [x1, y1, x2, y2],
                'center': [cx, cy],
                'class_name': class_name,
                'confidence': float(box.conf[0])
            })

        return detections

    def draw_detections(self, image, detections):
        """在图像上绘制检测结果"""
        img_copy = image.copy()

        for det in detections:
            x1, y1, x2, y2 = det['bbox']
            cx, cy = det['center']

            # 绘制边界框
            cv2.rectangle(img_copy, (x1, y1), (x2, y2), (0, 255, 0), 2)

            # 绘制中心点
            cv2.circle(img_copy, (cx, cy), 5, (0, 0, 255), -1)

            # 绘制标签
            label = f"{det['class_name']}: {det['confidence']:.2f}"
            cv2.putText(img_copy, label, (x1, y1 - 10),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

        return img_copy


# ============================================================
# 使用示例
# ============================================================

# 初始化检测器
detector = ObjectDetector(model_path='yolov8m.pt', conf_threshold=0.5)

# 读取图像
image = cv2.imread('test.jpg')

# 检测特定目标（如：只检测瓶子和杯子）
detections = detector.detect(image, target_classes=['bottle', 'cup'])

# 打印结果
for det in detections:
    print(f"检测到 {det['class_name']}，位置: {det['center']}，置信度: {det['confidence']:.2f}")

# 可视化
result_image = detector.draw_detections(image, detections)
cv2.imshow('Detections', result_image)
cv2.waitKey(0)
```

## 2️⃣ 3D位置计算模块

### 2.1 像素坐标到3D坐标的转换

```
┌─────────────────────────────────────────────────────────────┐
│                   3D位置计算原理                             │
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  【相机成像模型】                                            │
│                                                              │
│       3D世界点 P(X,Y,Z)                                     │
│              ↓                                               │
│         相机投影                                             │
│              ↓                                               │
│       2D像素点 p(u,v)                                       │
│                                                              │
│  【逆过程：2D → 3D】                                        │
│                                                              │
│       2D像素点 p(u,v) + 深度 d                              │
│              ↓                                               │
│         反投影                                               │
│              ↓                                               │
│       3D相机坐标 P(X,Y,Z)                                   │
│                                                              │
│  【计算公式】                                                │
│  ┌─────────────────────────────────────────────────────┐   │
│  │                                                       │   │
│  │   X = (u - cx) × d / fx                              │   │
│  │   Y = (v - cy) × d / fy                              │   │
│  │   Z = d                                               │   │
│  │                                                       │   │
│  │   其中：                                              │   │
│  │   - (u, v): 像素坐标                                 │   │
│  │   - d: 深度值（米）                                  │   │
│  │   - (cx, cy): 图像主点（通常是图像中心）             │   │
│  │   - (fx, fy): 焦距（像素单位）                       │   │
│  │                                                       │   │
│  │   相机内参矩阵 K:                                     │   │
│  │   ┌           ┐                                       │   │
│  │   │ fx  0  cx │                                       │   │
│  │   │ 0  fy  cy │                                       │   │
│  │   │ 0   0   1 │                                       │   │
│  │   └           ┘                                       │   │
│  │                                                       │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                              │
└─────────────────────────────────────────────────────────────┘
```

### 2.2 代码实现

```python
# ============================================================
# 3D位置计算模块 - 使用 Intel RealSense
# ============================================================

import pyrealsense2 as rs
import numpy as np

class DepthCamera:
    """深度相机封装类"""

    def __init__(self):
        """初始化RealSense相机"""
        self.pipeline = rs.pipeline()
        self.config = rs.config()

        # 配置RGB和深度流
        self.config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
        self.config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)

        # 启动相机
        self.profile = self.pipeline.start(self.config)

        # 获取深度传感器的内参
        depth_stream = self.profile.get_stream(rs.stream.depth)
        self.intrinsics = depth_stream.as_video_stream_profile().get_intrinsics()

        # 创建对齐对象（将深度图对齐到彩色图）
        self.align = rs.align(rs.stream.color)

        print(f"相机内参: fx={self.intrinsics.fx}, fy={self.intrinsics.fy}")
        print(f"         cx={self.intrinsics.ppx}, cy={self.intrinsics.ppy}")

    def get_frames(self):
        """获取RGB和深度帧"""
        frames = self.pipeline.wait_for_frames()

        # 对齐深度帧到彩色帧
        aligned_frames = self.align.process(frames)

        color_frame = aligned_frames.get_color_frame()
        depth_frame = aligned_frames.get_depth_frame()

        if not color_frame or not depth_frame:
            return None, None, None

        color_image = np.asanyarray(color_frame.get_data())
        depth_image = np.asanyarray(depth_frame.get_data())

        return color_image, depth_image, depth_frame

    def pixel_to_3d(self, u, v, depth_frame):
        """
        将像素坐标转换为相机坐标系下的3D点

        Args:
            u, v: 像素坐标
            depth_frame: RealSense深度帧

        Returns:
            [X, Y, Z]: 相机坐标系下的3D坐标（米）
        """
        # 获取该像素点的深度值（米）
        depth = depth_frame.get_distance(int(u), int(v))

        if depth == 0:
            return None  # 深度无效

        # 使用RealSense API进行反投影
        point_3d = rs.rs2_deproject_pixel_to_point(
            self.intrinsics,
            [u, v],
            depth
        )

        return point_3d  # [X, Y, Z] 单位：米

    def get_depth_at_pixel(self, u, v, depth_frame, window_size=5):
        """
        获取像素点的深度值（使用窗口平均减少噪声）

        Args:
            u, v: 像素坐标
            depth_frame: 深度帧
            window_size: 窗口大小

        Returns:
            平均深度值（米）
        """
        depths = []
        half = window_size // 2

        for du in range(-half, half + 1):
            for dv in range(-half, half + 1):
                d = depth_frame.get_distance(int(u + du), int(v + dv))
                if d > 0:
                    depths.append(d)

        if not depths:
            return 0

        return np.median(depths)  # 使用中值更鲁棒

    def release(self):
        """释放相机资源"""
        self.pipeline.stop()


class PositionCalculator:
    """3D位置计算器"""

    def __init__(self, camera):
        """
        Args:
            camera: DepthCamera实例
        """
        self.camera = camera

    def calculate_object_position(self, detection, depth_frame):
        """
        计算检测到的物体的3D位置

        Args:
            detection: 检测结果字典，包含 'center' 键
            depth_frame: 深度帧

        Returns:
            dict: 包含3D位置信息
        """
        cx, cy = detection['center']

        # 获取深度（使用窗口平均）
        depth = self.camera.get_depth_at_pixel(cx, cy, depth_frame, window_size=5)

        if depth == 0:
            return {
                'success': False,
                'error': '无法获取有效深度'
            }

        # 计算3D位置
        point_3d = self.camera.pixel_to_3d(cx, cy, depth_frame)

        return {
            'success': True,
            'position_camera': point_3d,  # 相机坐标系
            'depth': depth,
            'pixel': [cx, cy]
        }


# ============================================================
# 使用示例
# ============================================================

# 初始化
camera = DepthCamera()
detector = ObjectDetector('yolov8m.pt')
position_calc = PositionCalculator(camera)

# 主循环
while True:
    # 获取图像
    color_image, depth_image, depth_frame = camera.get_frames()
    if color_image is None:
        continue

    # 目标检测
    detections = detector.detect(color_image, target_classes=['bottle'])

    # 计算3D位置
    for det in detections:
        result = position_calc.calculate_object_position(det, depth_frame)

        if result['success']:
            pos = result['position_camera']
            print(f"目标 {det['class_name']} 的相机坐标: "
                  f"X={pos[0]:.3f}m, Y={pos[1]:.3f}m, Z={pos[2]:.3f}m")

    # 显示
    cv2.imshow('Color', detector.draw_detections(color_image, detections))
    if cv2.waitKey(1) == ord('q'):
        break

camera.release()
```

## 3️⃣ 手眼标定模块

### 3.1 原理说明

```
┌─────────────────────────────────────────────────────────────┐
│                      手眼标定原理                            │
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  【问题】                                                    │
│  相机看到的坐标（相机坐标系）≠ 机械臂的坐标（基座坐标系）   │
│  需要找到两个坐标系之间的变换关系                           │
│                                                              │
│  【数学表示】                                                │
│  ┌─────────────────────────────────────────────────────┐   │
│  │                                                       │   │
│  │   P_base = T_cam_to_base × P_camera                  │   │
│  │                                                       │   │
│  │   其中：                                              │   │
│  │   - P_camera: 相机坐标系下的点                       │   │
│  │   - P_base: 机械臂基座坐标系下的点                   │   │
│  │   - T_cam_to_base: 4×4变换矩阵（要求的）             │   │
│  │                                                       │   │
│  │   变换矩阵结构：                                      │   │
│  │   T = ┌         ┐                                     │   │
│  │       │ R   t   │   R: 3×3旋转矩阵                   │   │
│  │       │ 0   1   │   t: 3×1平移向量                   │   │
│  │       └         ┘                                     │   │
│  │                                                       │   │
│  └─────────────────────────────────────────────────────┘   │
│                                                              │
│  【标定方法：Eye-to-Hand】                                  │
│                                                              │
│       📷 相机（固定）                                       │
│        │                                                     │
│        │  观测标定板                                        │
│        ↓                                                     │
│       ┌───────┐                                             │
│       │棋盘格 │ ← 固定在机械臂末端                         │
│       └───┬───┘                                             │
│           │                                                  │
│         🦾 机械臂                                           │
│                                                              │
│  步骤：                                                      │
│  1. 在机械臂末端固定棋盘格标定板                            │
│  2. 移动机械臂到N个不同位置（10-20个）                     │

# ============================================================
# 手眼标定模块
# ============================================================

import cv2
import numpy as np
from scipy.spatial.transform import Rotation

class HandEyeCalibrator:
    """手眼标定器"""

    def __init__(self, board_size=(9, 6), square_size=0.025):
        """
        Args:
            board_size: 棋盘格内角点数量 (列, 行)
            square_size: 每个方格的边长（米）
        """
        self.board_size = board_size
        self.square_size = square_size

        # 生成棋盘格3D点
        self.objp = np.zeros((board_size[0] * board_size[1], 3), np.float32)
        self.objp[:, :2] = np.mgrid[0:board_size[0], 0:board_size[1]].T.reshape(-1, 2)
        self.objp *= square_size

        # 存储标定数据
        self.R_gripper2base_list = []  # 机械臂末端到基座的旋转
        self.t_gripper2base_list = []  # 机械臂末端到基座的平移
        self.R_target2cam_list = []    # 标定板到相机的旋转
        self.t_target2cam_list = []    # 标定板到相机的平移

    def detect_chessboard(self, image, camera_matrix, dist_coeffs):
        """
        检测棋盘格并计算其相对于相机的位姿

        Args:
            image: BGR图像
            camera_matrix: 相机内参矩阵
            dist_coeffs: 畸变系数

        Returns:
            成功返回 (R, t)，失败返回 None
        """
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        # 检测棋盘格角点
        ret, corners = cv2.findChessboardCorners(gray, self.board_size, None)

        if not ret:
            return None

        # 亚像素精化
        criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
        corners = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)

        # PnP求解位姿
        ret, rvec, tvec = cv2.solvePnP(
            self.objp, corners, camera_matrix, dist_coeffs
        )

        if not ret:
            return None

        # 旋转向量转旋转矩阵
        R, _ = cv2.Rodrigues(rvec)
        t = tvec.flatten()

        return R, t

    def add_sample(self, R_gripper2base, t_gripper2base, R_target2cam, t_target2cam):
        """
        添加一组标定样本

        Args:
            R_gripper2base: 机械臂末端到基座的旋转矩阵 (3×3)
            t_gripper2base: 机械臂末端到基座的平移向量 (3,)
            R_target2cam: 标定板到相机的旋转矩阵 (3×3)
            t_target2cam: 标定板到相机的平移向量 (3,)
        """
        self.R_gripper2base_list.append(R_gripper2base)
        self.t_gripper2base_list.append(t_gripper2base.reshape(3, 1))
        self.R_target2cam_list.append(R_target2cam)
        self.t_target2cam_list.append(t_target2cam.reshape(3, 1))

        print(f"已添加样本，当前样本数: {len(self.R_gripper2base_list)}")

    def calibrate(self, method=cv2.CALIB_HAND_EYE_TSAI):
        """
        执行手眼标定

        Args:
            method: 标定方法，可选：
                - cv2.CALIB_HAND_EYE_TSAI (推荐)
                - cv2.CALIB_HAND_EYE_PARK
                - cv2.CALIB_HAND_EYE_HORAUD
                - cv2.CALIB_HAND_EYE_ANDREFF
                - cv2.CALIB_HAND_EYE_DANIILIDIS

        Returns:
            T_cam2base: 4×4变换矩阵（相机到基座）
        """
        if len(self.R_gripper2base_list) < 3:
            raise ValueError("至少需要3组样本进行标定")

        print(f"开始标定，使用 {len(self.R_gripper2base_list)} 组样本...")

        # OpenCV手眼标定
        R_cam2gripper, t_cam2gripper = cv2.calibrateHandEye(
            self.R_gripper2base_list,
            self.t_gripper2base_list,
            self.R_target2cam_list,
            self.t_target2cam_list,
            method=method
        )

        # 构建4×4变换矩阵
        T_cam2gripper = np.eye(4)
        T_cam2gripper[:3, :3] = R_cam2gripper
        T_cam2gripper[:3, 3] = t_cam2gripper.flatten()

        print("标定完成！")
        print(f"相机到末端的变换矩阵:\n{T_cam2gripper}")

        return T_cam2gripper

    def save_calibration(self, T_cam2gripper, filepath='hand_eye_calibration.npy'):
        """保存标定结果"""
        np.save(filepath, T_cam2gripper)
        print(f"标定结果已保存到: {filepath}")

    def load_calibration(self, filepath='hand_eye_calibration.npy'):
        """加载标定结果"""
        return np.load(filepath)


class CoordinateTransformer:
    """坐标变换器"""

    def __init__(self, T_cam2base):
        """
        Args:
            T_cam2base: 4×4变换矩阵（相机坐标系到机械臂基座坐标系）
        """
        self.T_cam2base = T_cam2base

    def camera_to_base(self, point_camera):
        """
        将相机坐标系下的点转换到机械臂基座坐标系

        Args:
            point_camera: [X, Y, Z] 相机坐标系下的点

        Returns:
            [X, Y, Z] 基座坐标系下的点
        """
        # 转为齐次坐标
        p_cam = np.array([point_camera[0], point_camera[1], point_camera[2], 1])

        # 变换
        p_base = self.T_cam2base @ p_cam

        return p_base[:3]

                print(f"样本数不足，当前: {sample_count}，至少需要: 10")

        elif key == ord('q'):
            break

    cv2.destroyAllWindows()

import rclpy

from geometry_msgs.msg import Pose, PoseStamped
from moveit_msgs.msg import MoveItErrorCodes
from pymoveit2 import MoveIt2
from pymoveit2.robots import ur5  # 根据您的机械臂型号选择

class ArmController(Node):
    """机械臂控制器"""

    def __init__(self):
        super().__init__('arm_controller')

        # 初始化MoveIt2
        self.moveit2 = MoveIt2(
            node=self,
            joint_names=ur5.joint_names(),          # 关节名称
            base_link_name=ur5.base_link_name(),    # 基座链接
            end_effector_name=ur5.end_effector_name(),  # 末端链接
            group_name="manipulator"                 # 规划组名称
        )

        # 设置规划参数
        self.moveit2.max_velocity = 0.5   # 最大速度比例
        self.moveit2.max_acceleration = 0.5  # 最大加速度比例

        self.get_logger().info("机械臂控制器初始化完成")

    def move_to_pose(self, x, y, z, qx=0.0, qy=0.707, qz=0.0, qw=0.707):
        """
        移动到目标位姿

        Args:
            x, y, z: 目标位置（米）
            qx, qy, qz, qw: 目标姿态（四元数）
                           默认值表示末端朝下

        Returns:
            bool: 是否成功
        """
        # 创建目标位姿
        target_pose = Pose()
        target_pose.position.x = x
        target_pose.position.y = y
        target_pose.position.z = z
        target_pose.orientation.x = qx
        target_pose.orientation.y = qy
        target_pose.orientation.z = qz
        target_pose.orientation.w = qw

        self.get_logger().info(f"规划移动到: ({x:.3f}, {y:.3f}, {z:.3f})")

        # 执行移动
        self.moveit2.move_to_pose(pose=target_pose)
        success = self.moveit2.wait_until_executed()

        if success:
            self.get_logger().info("移动完成")
        else:
            self.get_logger().error("移动失败")

        return success

    def move_to_joint_positions(self, joint_positions):
        """
        移动到指定关节角度

        Args:
            joint_positions: 关节角度列表 [θ1, θ2, θ3, θ4, θ5, θ6] (弧度)
        """
        self.moveit2.move_to_configuration(joint_positions)
        return self.moveit2.wait_until_executed()

    def move_home(self):
        """移动到初始位置"""
        home_position = [0.0, -1.57, 1.57, -1.57, -1.57, 0.0]  # 示例值
        return self.move_to_joint_positions(home_position)

    def get_current_pose(self):
        """获取当前末端位姿"""
        return self.moveit2.get_current_pose()

    def execute_cartesian_path(self, waypoints, step=0.01):
        """
        执行笛卡尔路径（直线运动）

        Args:
            waypoints: 路径点列表 [(x1,y1,z1), (x2,y2,z2), ...]
            step: 插值步长（米）
        """
        pose_list = []
        for wp in waypoints:
            pose = Pose()
            pose.position.x = wp[0]
            pose.position.y = wp[1]
            pose.position.z = wp[2]
            pose.orientation.x = 0.0
            pose.orientation.y = 0.707
            pose.orientation.z = 0.0
            pose.orientation.w = 0.707
            pose_list.append(pose)

        # 规划笛卡尔路径
        trajectory, fraction = self.moveit2.plan_cartesian_path(
            poses=pose_list,
            max_step=step,
            avoid_collisions=True
        )

        if fraction < 0.9:
            self.get_logger().warning(f"只能规划 {fraction*100:.1f}% 的路径")
            return False

        # 执行轨迹
        self.moveit2.execute(trajectory)
        return self.moveit2.wait_until_executed()

        if not detections:
            return {'success': False, 'error': '未检测到目标'}

        # 取置信度最高的检测结果
        best_detection = max(detections, key=lambda x: x['confidence'])

        # 计算3D位置（相机坐标系）
        position_result = self.position_calc.calculate_object_position(
            best_detection, depth_frame
        )

        if not position_result['success']:
            return position_result

        # 转换到机械臂基座坐标系
        position_camera = position_result['position_camera']
        position_base = self.transformer.camera_to_base(position_camera)

        return {
            'success': True,
            'class_name': best_detection['class_name'],
            'confidence': best_detection['confidence'],
            'position_camera': position_camera,
            'position_base': position_base,
            'image': color_image,
            'detection': best_detection
        }

    def grasp(self, target_position):
        """
        执行抓取动作

        Args:
            target_position: [x, y, z] 目标在基座坐标系下的位置（米）

        Returns:
            bool: 是否抓取成功
        """
        x, y, z = target_position

        # 转换为毫米（如果机械臂API使用毫米）
        x_mm = x * 1000
        y_mm = y * 1000
        z_mm = z * 1000

        print(f"目标位置: X={x_mm:.1f}mm, Y={y_mm:.1f}mm, Z={z_mm:.1f}mm")

        try:
            # 1. 移动到预抓取位置（目标正上方）
            pre_grasp_z = z_mm + self.pre_grasp_height * 1000
            print(f"步骤1: 移动到预抓取位置 (高度: {pre_grasp_z:.1f}mm)")
            self.arm.move_to_coords(x_mm, y_mm, pre_grasp_z, rx=180, ry=0, rz=0)

            # 2. 打开夹爪
            print("步骤2: 打开夹爪")
            self.arm.open_gripper()
            time.sleep(0.5)

            # 3. 下降到抓取位置
            grasp_z = z_mm + self.grasp_height_offset * 1000
            print(f"步骤3: 下降到抓取位置 (高度: {grasp_z:.1f}mm)")
            self.arm.move_to_coords(x_mm, y_mm, grasp_z, rx=180, ry=0, rz=0)
            time.sleep(0.5)

            # 4. 闭合夹爪
            print("步骤4: 闭合夹爪")
            self.arm.close_gripper()
            time.sleep(0.5)

            # 5. 检查是否抓取成功
            if self.arm.is_gripper_holding():
                print("✅ 检测到物体，抓取成功！")

                # 6. 提起物体
                lift_z = z_mm + self.lift_height * 1000
                print(f"步骤5: 提起物体 (高度: {lift_z:.1f}mm)")
                self.arm.move_to_coords(x_mm, y_mm, lift_z, rx=180, ry=0, rz=0)

                return True
            else:
                print("❌ 未检测到物体，抓取失败")
                # 打开夹爪并抬起
                self.arm.open_gripper()
                self.arm.move_to_coords(x_mm, y_mm, pre_grasp_z, rx=180, ry=0, rz=0)
                return False

        except Exception as e:
            print(f"抓取过程出错: {e}")
            return False

    def place(self, target_position):
        """
        放置物体

        Args:
            target_position: [x, y, z] 放置位置
        """
        x, y, z = target_position
        x_mm, y_mm, z_mm = x * 1000, y * 1000, z * 1000

        # 移动到放置位置上方
        self.arm.move_to_coords(x_mm, y_mm, z_mm + 100, rx=180, ry=0, rz=0)

        # 下降
        self.arm.move_to_coords(x_mm, y_mm, z_mm + 20, rx=180, ry=0, rz=0)

        # 打开夹爪
        self.arm.open_gripper()
        time.sleep(0.5)

        # 抬起
        self.arm.move_to_coords(x_mm, y_mm, z_mm + 100, rx=180, ry=0, rz=0)

    def run_grasp_task(self, target_class=None, place_position=None):
        """
        运行完整的抓取任务

        Args:
            target_class: 目标类别（可选）
            place_position: 放置位置（可选）

        Returns:
            bool: 任务是否成功
        """
        print("\n" + "="*50)
        print("开始抓取任务")
        print("="*50)

        # 步骤1: 检测目标
        print("\n[1/4] 检测目标...")
        result = self.detect_and_locate(target_class)

        if not result['success']:
            print(f"检测失败: {result.get('error', '未知错误')}")
            return False

        print(f"检测到: {result['class_name']} (置信度: {result['confidence']:.2f})")
        print(f"基座坐标: {result['position_base']}")

        # 显示检测结果
        cv2.imshow('Detection', self.detector.draw_detections(
            result['image'], [result['detection']]
        ))
        cv2.waitKey(1000)

        # 步骤2: 执行抓取
        print("\n[2/4] 执行抓取...")
        grasp_success = self.grasp(result['position_base'])

        if not grasp_success:
            print("抓取失败")
            return False

        # 步骤3: 放置（如果指定了放置位置）
        if place_position:
            print("\n[3/4] 搬运到目标位置...")
            self.place(place_position)
            print("放置完成")

        # 步骤4: 返回初始位置
        print("\n[4/4] 返回初始位置...")
        self.arm.move_to_angles([0, 0, 0, 0, 0, 0])  # 根据实际调整

        print("\n" + "="*50)
        print("✅ 抓取任务完成！")
        print("="*50)

        return True

    def release(self):
        """释放资源"""
        self.camera.release()
        cv2.destroyAllWindows()


# ============================================================
# 主程序入口
# ============================================================

def main():
    """主函数"""

    # 创建抓取系统
    system = AutoGraspingSystem(calibration_file='hand_eye_calibration.npy')

    try:
        # 运行抓取任务
        # 示例：抓取瓶子并放置到指定位置
        success = system.run_grasp_task(
            target_class='bottle',
            place_position=[0.3, 0.0, 0.1]  # 放置位置 (米)
        )

        if success:
            print("任务成功完成！")
        else:
            print("任务执行失败")

    except KeyboardInterrupt:
        print("\n用户中断")
    finally:
        system.release()


if __name__ == '__main__':
    main()
