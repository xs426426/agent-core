### 1.3 关键代码示例

```python
# ============================================================
# 无人机ROS 2控制节点
# ============================================================

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped, TwistStamped
from mavros_msgs.msg import State
from mavros_msgs.srv import CommandBool, SetMode
from std_msgs.msg import String
import json

class DroneController(Node):
    """无人机控制节点"""

    def __init__(self):
        super().__init__('drone_controller')

        # 状态变量
        self.current_state = State()
        self.current_pose = None

        # 订阅者
        self.state_sub = self.create_subscription(
            State, '/mavros/state', self.state_callback, 10)
        self.pose_sub = self.create_subscription(
            PoseStamped, '/mavros/local_position/pose', self.pose_callback, 10)

        # 发布者
        self.local_pos_pub = self.create_publisher(
            PoseStamped, '/mavros/setpoint_position/local', 10)
        self.target_pub = self.create_publisher(
            String, '/drone/target_found', 10)  # 发送给小车

        # 服务客户端
        self.arming_client = self.create_client(
            CommandBool, '/mavros/cmd/arming')
        self.set_mode_client = self.create_client(
            SetMode, '/mavros/set_mode')

        self.get_logger().info('无人机控制节点已启动')

    def state_callback(self, msg):
        self.current_state = msg

    def pose_callback(self, msg):
        self.current_pose = msg.pose

    async def arm(self):
        """解锁电机"""
        req = CommandBool.Request()
        req.value = True
        future = self.arming_client.call_async(req)
        await future
        return future.result().success

    async def set_mode(self, mode):
        """设置飞行模式"""
        req = SetMode.Request()
        req.custom_mode = mode
        future = self.set_mode_client.call_async(req)
        await future
        return future.result().mode_sent

    def fly_to_position(self, x, y, z):
        """飞行到指定位置"""
        pose = PoseStamped()
        pose.header.stamp = self.get_clock().now().to_msg()
        pose.header.frame_id = 'map'
        pose.pose.position.x = x
        pose.pose.position.y = y
        pose.pose.position.z = z
        pose.pose.orientation.w = 1.0
        self.local_pos_pub.publish(pose)

    def publish_target_position(self, target_x, target_y, target_z):
        """发布目标位置给小车"""
        msg = String()
        msg.data = json.dumps({
            'target_position': {
                'x': target_x,
                'y': target_y,
                'z': target_z
            },
            'timestamp': self.get_clock().now().nanoseconds
        })
        self.target_pub.publish(msg)
        self.get_logger().info(f'已发送目标位置: ({target_x}, {target_y}, {target_z})')


# ============================================================
# 目标搜索节点
# ============================================================

class TargetSearcher(Node):
    """目标搜索节点 - 集成YOLO检测"""

    def __init__(self):
        super().__init__('target_searcher')

        from ultralytics import YOLO
        import pyrealsense2 as rs
        import numpy as np

        # 初始化YOLO
        self.detector = YOLO('yolov8m.pt')

        # 初始化RealSense
        self.pipeline = rs.pipeline()
        config = rs.config()
        config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
        config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
        self.pipeline.start(config)

        # 发布检测结果
        self.detection_pub = self.create_publisher(
            String, '/drone/detection', 10)

        # 定时器 - 30Hz检测
        self.timer = self.create_timer(0.033, self.detection_loop)

    def detection_loop(self):
        """检测循环"""
        import numpy as np

        frames = self.pipeline.wait_for_frames()
        color_frame = frames.get_color_frame()
        depth_frame = frames.get_depth_frame()

        if not color_frame or not depth_frame:
            return

        color_image = np.asanyarray(color_frame.get_data())

        # YOLO检测
        results = self.detector(color_image, conf=0.5)

        for box in results[0].boxes:
            class_name = self.detector.names[int(box.cls)]
            x1, y1, x2, y2 = map(int, box.xyxy[0])
            cx, cy = (x1 + x2) // 2, (y1 + y2) // 2

            # 获取深度
            depth = depth_frame.get_distance(cx, cy)

            if depth > 0:
                # 发布检测结果
                msg = String()
                msg.data = json.dumps({
                    'class': class_name,
                    'confidence': float(box.conf),
                    'pixel': [cx, cy],
                    'depth': depth
                })
                self.detection_pub.publish(msg)
```

```python
# ============================================================
# 协同控制器 - 管理无人机和小车的协同工作
# ============================================================

import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient, ActionServer
from rclpy.callback_groups import ReentrantCallbackGroup
from std_msgs.msg import String
from geometry_msgs.msg import PoseStamped
import json
from enum import Enum

class MissionPhase(Enum):
    IDLE = "idle"
    DRONE_TAKEOFF = "drone_takeoff"
    DRONE_SEARCH = "drone_search"
    TARGET_FOUND = "target_found"
    CAR_NAVIGATING = "car_navigating"
    CAR_ARRIVED = "car_arrived"
    ARM_GRASPING = "arm_grasping"
    DRONE_RETURNING = "drone_returning"
    DRONE_LANDING = "drone_landing"
    MISSION_COMPLETE = "mission_complete"
    ERROR = "error"


class CoordinationController(Node):
    """空地协同控制器"""

    def __init__(self):
        super().__init__('coordination_controller')

        self.callback_group = ReentrantCallbackGroup()

        # 当前任务状态
        self.current_phase = MissionPhase.IDLE
        self.target_position = None

        # 订阅各系统状态
        self.drone_state_sub = self.create_subscription(
            String, '/drone/state', self.drone_state_callback, 10)
        self.drone_target_sub = self.create_subscription(
            String, '/drone/target_found', self.target_found_callback, 10)
        self.car_status_sub = self.create_subscription(
            String, '/car/status', self.car_status_callback, 10)
        self.arm_status_sub = self.create_subscription(
            String, '/arm/status', self.arm_status_callback, 10)

        # 发布协同指令
        self.drone_cmd_pub = self.create_publisher(String, '/drone/command', 10)
        self.car_cmd_pub = self.create_publisher(String, '/car/command', 10)
        self.arm_cmd_pub = self.create_publisher(String, '/arm/command', 10)
        self.mission_status_pub = self.create_publisher(String, '/coordination/mission_status', 10)

        # 服务：启动任务
        self.start_mission_srv = self.create_service(
            Trigger, '/coordination/start_mission', self.start_mission_callback)

        self.get_logger().info('协同控制器已启动')

    def start_mission_callback(self, request, response):
        """启动协同任务"""
        if self.current_phase != MissionPhase.IDLE:
            response.success = False
            response.message = f'任务进行中: {self.current_phase.value}'
            return response

        self.get_logger().info('启动空地协同任务')
        self.execute_mission()

        response.success = True
        response.message = '任务已启动'
        return response

    def execute_mission(self):
        """执行完整任务流程"""
        # 阶段1: 无人机起飞
        self.set_phase(MissionPhase.DRONE_TAKEOFF)
        self.send_drone_command('takeoff', {'altitude': 2.0})

    def drone_state_callback(self, msg):
        """处理无人机状态"""
        state = json.loads(msg.data)

        if self.current_phase == MissionPhase.DRONE_TAKEOFF:
            if state.get('altitude', 0) >= 1.8:  # 接近目标高度
                # 开始搜索
                self.set_phase(MissionPhase.DRONE_SEARCH)
                self.send_drone_command('search', {
                    'pattern': 'spiral',
                    'target_class': 'box'
                })

    def target_found_callback(self, msg):
        """处理目标发现事件"""
        if self.current_phase != MissionPhase.DRONE_SEARCH:
            return

        data = json.loads(msg.data)
        self.target_position = data['target_position']

        self.get_logger().info(f'目标已发现: {self.target_position}')
        self.set_phase(MissionPhase.TARGET_FOUND)

        # 指令小车导航到目标
        self.set_phase(MissionPhase.CAR_NAVIGATING)
        self.send_car_command('navigate', {
            'x': self.target_position['x'],
            'y': self.target_position['y']
        })

        # 无人机进入跟随/待命模式
        self.send_drone_command('hover', {})

    def car_status_callback(self, msg):
        """处理小车状态"""
        status = json.loads(msg.data)

        if self.current_phase == MissionPhase.CAR_NAVIGATING:
            if status.get('status') == 'arrived':
                self.set_phase(MissionPhase.CAR_ARRIVED)

                # 启动机械臂抓取
                self.set_phase(MissionPhase.ARM_GRASPING)
                self.send_arm_command('grasp', {
                    'x': self.target_position['x'],
                    'y': self.target_position['y'],
                    'z': self.target_position['z']
                })

    def arm_status_callback(self, msg):
        """处理机械臂状态"""
        status = json.loads(msg.data)

        if self.current_phase == MissionPhase.ARM_GRASPING:
            if status.get('status') == 'grasp_complete':
                self.get_logger().info('抓取完成')

                # 无人机返回降落
                self.set_phase(MissionPhase.DRONE_RETURNING)
                self.send_drone_command('return_to_car', {})

    def set_phase(self, phase: MissionPhase):
        """设置任务阶段"""
        self.current_phase = phase
        self.get_logger().info(f'任务阶段: {phase.value}')

        # 发布任务状态
        status_msg = String()
        status_msg.data = json.dumps({
            'phase': phase.value,
            'timestamp': self.get_clock().now().nanoseconds
        })
        self.mission_status_pub.publish(status_msg)

    def send_drone_command(self, command: str, params: dict):
        """发送无人机指令"""
        msg = String()
        msg.data = json.dumps({'command': command, 'params': params})
        self.drone_cmd_pub.publish(msg)

    def send_car_command(self, command: str, params: dict):
        """发送小车指令"""
        msg = String()
        msg.data = json.dumps({'command': command, 'params': params})
        self.car_cmd_pub.publish(msg)

    def send_arm_command(self, command: str, params: dict):
        """发送机械臂指令"""
        msg = String()
        msg.data = json.dumps({'command': command, 'params': params})
        self.arm_cmd_pub.publish(msg)


def main():
    rclpy.init()
    node = CoordinationController()
    rclpy.spin(node)
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

```python
        self.landing_status_pub = self.create_publisher(
            Bool, '/drone/landing_complete', 10)

        # 控制定时器 (50Hz)
        self.control_timer = self.create_timer(0.02, self.control_loop)

        # 当前检测结果
        self.marker_pose = None

        self.get_logger().info('精准降落控制器已启动')

    def image_callback(self, msg):
        """处理下视相机图像"""
        if not self.is_landing:
            return

        # 转换图像
        image = self.bridge.imgmsg_to_cv2(msg, 'bgr8')
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        # 检测AprilTag
        corners, ids, rejected = self.detector.detectMarkers(gray)

        if ids is not None and len(ids) > 0:
            # 估计标记位姿
            rvecs, tvecs, _ = aruco.estimatePoseSingleMarkers(
                corners, self.marker_size, self.camera_matrix, self.dist_coeffs)

            # 取第一个检测到的标记
            tvec = tvecs[0][0]

            self.marker_pose = {
                'x': tvec[0],  # 横向偏移（米）
                'y': tvec[1],  # 纵向偏移（米）
                'z': tvec[2]   # 距离（米）
            }
            self.target_detected = True

            self.get_logger().debug(
                f'标记位置: x={tvec[0]:.3f}, y={tvec[1]:.3f}, z={tvec[2]:.3f}')
        else:
            self.target_detected = False

    def height_callback(self, msg):
        """处理测距仪数据"""
        self.current_height = msg.range

    def control_loop(self):
        """控制循环"""
        if not self.is_landing:
            return

        vel_msg = TwistStamped()
        vel_msg.header.stamp = self.get_clock().now().to_msg()

        if self.target_detected and self.marker_pose:
            # 计算误差
            error_x = self.marker_pose['x']
            error_y = self.marker_pose['y']

            # PID控制
            # P项
            vx = -self.kp_xy * error_x
            vy = -self.kp_xy * error_y

            # I项
            self.error_integral_x += error_x * 0.02
            self.error_integral_y += error_y * 0.02
            vx += -self.ki_xy * self.error_integral_x
            vy += -self.ki_xy * self.error_integral_y

            # D项
            vx += -self.kd_xy * (error_x - self.last_error_x) / 0.02
            vy += -self.kd_xy * (error_y - self.last_error_y) / 0.02

            self.last_error_x = error_x
            self.last_error_y = error_y

            # 限制速度
            max_vel = 0.5  # m/s
            vx = np.clip(vx, -max_vel, max_vel)
            vy = np.clip(vy, -max_vel, max_vel)

            # 高度控制
            if self.current_height > 0.3:
                # 如果对准良好，下降
                if abs(error_x) < 0.1 and abs(error_y) < 0.1:
                    vz = -self.kp_z * self.current_height
                    vz = np.clip(vz, -0.3, 0)  # 最大下降速度0.3m/s
                else:
                    vz = 0  # 先对准再下降
            else:
                # 接近地面，缓慢下降
                vz = -0.1

            vel_msg.twist.linear.x = vx
            vel_msg.twist.linear.y = vy
            vel_msg.twist.linear.z = vz

            # 检查是否着陆
            if self.current_height < 0.1:
                self.complete_landing()
                return

        else:
            # 未检测到标记，悬停
            vel_msg.twist.linear.x = 0.0
            vel_msg.twist.linear.y = 0.0
            vel_msg.twist.linear.z = 0.0
            self.get_logger().warn('未检测到降落标记')

        self.velocity_pub.publish(vel_msg)

    def start_landing(self):
        """开始降落"""
        self.is_landing = True
        self.error_integral_x = 0.0
        self.error_integral_y = 0.0
        self.get_logger().info('开始精准降落')

    def complete_landing(self):
        """完成降落"""
        self.is_landing = False

        # 发送零速度
        vel_msg = TwistStamped()
        vel_msg.header.stamp = self.get_clock().now().to_msg()
        self.velocity_pub.publish(vel_msg)

        # 发布降落完成
        complete_msg = Bool()
        complete_msg.data = True
        self.landing_status_pub.publish(complete_msg)

        self.get_logger().info('降落完成！')
```
